index_from <- function(fit, tvar) { lev <- levels(fit$model[[tvar]]); cc <- coef(fit)
  th <- setNames(rep(0, length(lev)), lev); nm <- paste0(tvar, lev)
  hit <- nm %in% names(cc); th[hit] <- cc[nm[hit]]
  data.frame(period = lev, index = 100*exp(th - th[1]), row.names = NULL) }
band_from <- function(fit, tvar, cl) { lev <- levels(fit$model[[tvar]]); cc <- coef(fit)
  V <- vcovCL(fit, cluster = cl); nm <- paste0(tvar, lev)
  th <- se <- setNames(rep(0, length(lev)), lev); hit <- nm %in% names(cc)
  th[hit] <- cc[nm[hit]]; se[hit] <- sqrt(diag(V)[nm[hit]])
  data.frame(period = lev, index = 100*exp(th), lo = 100*exp(th-1.96*se), hi = 100*exp(th+1.96*se)) }

idx_ret <- function(ix) setNames(c(NA, diff(ix$index)/head(ix$index,-1)), ix$period)

## ---- 8. mixture robustness: common time, house and country effects ---------  # R2
ARI <- function(a,b) { t <- table(a,b); n <- sum(t); ch <- function(x) x*(x-1)/2
  si <- sum(ch(rowSums(t))); sj <- sum(ch(colSums(t))); sij <- sum(ch(t))
  e <- si*sj/ch(n); (sij-e)/(0.5*(si+sj)-e) }
relabel <- function(cl) { u <- sort(unique(cl))
  rl <- lapply(u, function(s) idx_ret(interp_index(reg_index(cl == s))))
  cx <- t(sapply(rl, comov)); j1 <- which.max(cx[,"SP500"]); j3 <- which.min(cx[,"SP500"])
  j2 <- setdiff(seq_along(u), c(j1,j3)); lb <- character(length(u))
  lb[j1] <- "R1"; lb[j2] <- "R2"; lb[j3] <- "R3"
  list(lab = lb[match(cl,u)], cm = cx, rets = rl, ord = c(j1,j2,j3), labs = lb, u = u) }
d$res1 <- residuals(lm(lnprice ~ qtr_f + house + country, data = d))
d$res2 <- residuals(lm(lnprice ~ qtr_f, data = d))
set.seed(1); fm1 <- stepFlexmix(as.formula(paste0("res1~", SEG_HED)), data = d, k = 3, nrep = 5,
  control = list(minprior = 0.05), verbose = FALSE)
set.seed(1); fm2 <- stepFlexmix(as.formula(paste0("res2~", SEG_HED, "+house+country")), data = d,
  k = 3, nrep = 5, control = list(minprior = 0.05), verbose = FALSE)
R0 <- list(lab = d$segn, cm = cm, rets = rl0, ord = c(i1,i2,i3), labs = labels3, u = segs)
RB <- list(M0 = R0, M1 = relabel(clusters(fm1)), M2 = relabel(clusters(fm2)))
mixture_robust <- do.call(rbind, lapply(names(RB), function(nm) { R <- RB[[nm]]
  k1 <- R$ord[1]; k2 <- R$ord[2]
  ix1 <- interp_index(reg_index(R$lab == "R1")); ix2 <- interp_index(reg_index(R$lab == "R2"))
  cl <- if (nm == "M0") d$seg else if (nm == "M1") clusters(fm1) else clusters(fm2)
  Pm <- if (nm == "M0") P0 else posterior(if (nm == "M1") fm1 else fm2)
  hp <- round(100*table(R$lab[d$POST == 0])/sum(d$POST == 0),1)
  hq <- round(100*table(R$lab[d$POST == 1])/sum(d$POST == 1),1)
  data.frame(spec = nm, sizes = paste(as.integer(table(cl)), collapse="/"),
    ARI_vs_M0 = round(ARI(d$seg, cl),3), mean_max_posterior = round(mean(apply(Pm,1,max)),3),
    share_post_gt_09 = round(mean(apply(Pm,1,max) > 0.9),3),
    R1_share_pre = hp["R1"], R1_share_post = hq["R1"],
    R1_rho_SP = round(R$cm[k1,"SP500"],3), R1_rho_K500 = round(R$cm[k1,"K500"],3),
    R1_end = round(tail(ix1$index,1),1), R2_end = round(tail(ix2$index,1),1),
    R1_vol = round(200*sd(R$rets[[k1]], na.rm=TRUE),2), R2_vol = round(200*sd(R$rets[[k2]], na.rm=TRUE),2),
    R1_price_med = median(d$price[R$lab == "R1"]), R1_pct30 = round(100*mean(d$age[R$lab=="R1"] >= 30),1),
    row.names = NULL) }))

inter_full <- function(sr, mcol, brk = "2021 Q1", excess = TRUE, mk = mkt_r) {
  jj <- merge(data.frame(qtr = names(sr), y = as.numeric(sr)), mk[, c("qtr",mcol,"rf")], by = "qtr")
  jj <- jj[complete.cases(jj), ]; jj$POST <- as.integer(jj$qtr >= brk)
  jj$yy <- if (excess) jj$y - jj$rf else jj$y
  jj$xx <- if (excess) jj[[mcol]] - jj$rf else jj[[mcol]]
  m <- lm(yy ~ xx + POST + xx:POST, data = jj); V <- vcovHC(m,"HC1")
  ct <- coeftest(m, vcov = V); ci <- coefci(m, vcov. = V)
  r1 <- cor(jj$y[jj$POST==0], jj[[mcol]][jj$POST==0]); r2 <- cor(jj$y[jj$POST==1], jj[[mcol]][jj$POST==1])
  z <- (atanh(r2)-atanh(r1))/sqrt(1/(sum(jj$POST==0)-3)+1/(sum(jj$POST==1)-3))
  data.frame(market = mcol, break_at = brk, n = nrow(jj), beta_pre = ct["xx",1],
    delta_beta = ct["xx:POST",1], se = ct["xx:POST",2], ci_low = ci["xx:POST",1],
    ci_high = ci["xx:POST",2], p = ct["xx:POST",4], rho_pre = r1, rho_post = r2,
    fisher_p = 2*pnorm(-abs(z)), row.names = NULL) }

## truncation at the US$50,000 cap
trunc_fit <- function(formula, data, cap) {
  mf <- model.frame(formula, data); y <- model.response(mf); X <- model.matrix(formula, mf)
  X <- X[, colSums(abs(X)) > 0, drop = FALSE]; q <- qr(X); X <- X[, q$pivot[seq_len(q$rank)], drop = FALSE]
  b0 <- qr.coef(qr(X), y); s0 <- log(sd(y - drop(X %*% b0)))
  nll <- function(th) { bb <- th[-length(th)]; s <- exp(th[length(th)]); mu <- drop(X %*% bb)
    -sum(dnorm(y, mu, s, log = TRUE) - pnorm((cap-mu)/s, log.p = TRUE)) }
  gr <- function(th) { bb <- th[-length(th)]; s <- exp(th[length(th)]); mu <- drop(X %*% bb)
    z <- (y-mu)/s; zc <- (cap-mu)/s; lam <- exp(dnorm(zc, log=TRUE) - pnorm(zc, log.p=TRUE))
    c(-drop(crossprod(X, (z+lam)/s)), -sum(z^2 - 1 + lam*zc)) }
  o <- optim(c(b0,s0), nll, gr, method = "BFGS", control = list(maxit = 10000, reltol = 1e-12))
  setNames(o$par[-length(o$par)], colnames(X)) }
trunc_index <- function(sub) { sub <- droplevels(sub)
  cc <- trunc_fit(as.formula(paste0(adapt(sub), "+qtr_f")), sub, log(50000))
  lev <- levels(sub$qtr_f); th <- setNames(rep(0, length(lev)), lev); nm <- paste0("qtr_f", lev)
  hit <- nm %in% names(cc); th[hit] <- cc[nm[hit]]
  data.frame(period = lev, index = 100*exp(th - th[1])) }
